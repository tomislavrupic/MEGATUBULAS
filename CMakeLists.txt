cmake_minimum_required(VERSION 3.24)
project(MEGATUBULAS VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(JUCE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/third_party/JUCE" CACHE PATH "JUCE 8.0.14 source")
if(NOT EXISTS "${JUCE_PATH}/CMakeLists.txt")
 message(FATAL_ERROR "Run scripts/bootstrap.sh, or pass -DJUCE_PATH=/path/to/JUCE-8.0.14")
endif()
add_subdirectory("${JUCE_PATH}" "${CMAKE_BINARY_DIR}/JUCE" EXCLUDE_FROM_ALL)
set(formats VST3 Standalone)
if(APPLE)
 list(APPEND formats AU)
endif()
juce_add_plugin(Megatubulas COMPANY_NAME "Pixel Records" BUNDLE_ID com.pixelrecords.megatubulas
 PLUGIN_MANUFACTURER_CODE PxRc PLUGIN_CODE Mctb FORMATS ${formats}
 PRODUCT_NAME MEGATUBULAS IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
 IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE
 VST3_CATEGORIES Fx Distortion AU_MAIN_TYPE kAudioUnitType_Effect)
file(GLOB MICRO_FRAMES CONFIGURE_DEPENDS "Artwork/Animation/v2/*.png")
juce_add_binary_data(MicroArt SOURCES ${MICRO_FRAMES} Artwork/AI/frame-background-v030.png Artwork/AI/lattice-strip-v2.png)
target_link_libraries(Megatubulas PRIVATE MicroArt)
target_sources(Megatubulas PRIVATE Source/Processor.cpp Source/Editor.cpp)
target_compile_definitions(Megatubulas PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0)
target_link_libraries(Megatubulas PUBLIC juce::juce_audio_utils juce::juce_dsp PRIVATE juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
juce_add_console_app(MicroTests PRODUCT_NAME MicroTests)
target_sources(MicroTests PRIVATE Tests/Tests.cpp)
target_include_directories(MicroTests PRIVATE Source)
target_compile_definitions(MicroTests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(MicroTests PRIVATE juce::juce_dsp)
juce_add_gui_app(MicroPreview PRODUCT_NAME MicroPreview)
target_sources(MicroPreview PRIVATE Tests/Preview.cpp)
target_link_libraries(MicroPreview PRIVATE Megatubulas)
enable_testing()
add_executable(MicroFMTests Tests/FMTests.cpp)
target_include_directories(MicroFMTests PRIVATE Source)
add_test(NAME harmonic_fm COMMAND MicroFMTests)
add_test(NAME numerical COMMAND MicroTests)
juce_add_gui_app(MicroIntegration PRODUCT_NAME MicroIntegration)
target_sources(MicroIntegration PRIVATE Tests/Integration.cpp)
target_link_libraries(MicroIntegration PRIVATE Megatubulas)
add_test(NAME integration COMMAND MicroIntegration)
juce_add_console_app(MegaRender PRODUCT_NAME MegaRender)
target_sources(MegaRender PRIVATE Tests/Render.cpp)
target_include_directories(MegaRender PRIVATE Source)
target_compile_definitions(MegaRender PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(MegaRender PRIVATE juce::juce_dsp juce::juce_audio_formats)

find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
 add_test(NAME renderer_cli COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/Tests/render_cli.py" "$<TARGET_FILE:MegaRender>")
endif()
juce_add_console_app(MicroFMPerformance PRODUCT_NAME MicroFMPerformance)
target_sources(MicroFMPerformance PRIVATE Tests/FMPerformance.cpp)
target_include_directories(MicroFMPerformance PRIVATE Source)
target_compile_definitions(MicroFMPerformance PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(MicroFMPerformance PRIVATE juce::juce_dsp)
